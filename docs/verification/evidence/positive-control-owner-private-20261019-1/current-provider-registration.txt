fn control_device_ioctl(cmd: u32, arg: usize) -> Result<isize> {
    match cmd {
        IHK_DEVICE_GET_BUILDID => compatibility_build_id(arg),
        _ => Err(EINVAL),
    }
}

            // instance becomes live; the raw scalar argument is never a pointer.

                ihk_os_create_unbooted_v10(
                    IHK_SMP_CONTROL_DEVICE_MINOR,
                    THIS_MODULE.as_ptr().cast(),
                    arg as u64,
                    1,
                    Some(ihk_smp_os_ioctl_v2),
                    Some(ihk_smp_os_release_v2),
                    Some(ihk_smp_prepare_boot_v3),
                    Some(ihk_smp_start_boot_v3),
                    Some(application_open),
                    Some(application_invoke),
                    Some(application_close),
                    Some(ihk_smp_shutdown_v6),
                    Some(application_adopt_fork),
                    Some(application_fork_progress),
                    Some(application_adopt_fork_scoped),
                    Some(application_prepare_fork_link),
                    Some(application_fork_progress_payload),
                    Some(application_prepare_fork_ack),
                    Some(application_fork_publication_verify),
                    Some(application_release_fork_ack),
                )
            };
            if result < 0 {
                Err(provider_status_error(result))
            } else {
                Ok(result as isize)
            }
