# NONSHIPPED selected-user-subtree qualification, not a full image build.
cmake_minimum_required(VERSION 3.11)
# Exact immutable SDK image tool already authenticated by the retained clang descriptor.
set(CMAKE_CXX_COMPILER "/usr/bin/clang" CACHE FILEPATH "Pinned SDK CXX driver")
set(CMAKE_CXX_COMPILER_ARG1 "--driver-mode=g++" CACHE STRING "Use pinned clang as CXX driver")
if(NOT CMAKE_CXX_COMPILER STREQUAL "/usr/bin/clang" OR NOT CMAKE_CXX_COMPILER_ARG1 STREQUAL "--driver-mode=g++")
  message(FATAL_ERROR "This qualification requires the pinned SDK clang CXX driver")
endif()
project(m15_bfd_selected_tools C CXX ASM)
if(NOT STAGED_USER_SOURCE OR NOT ORIGINAL_SOURCE)
  message(FATAL_ERROR "Explicit private staged user source and original source required")
endif()
set(CMAKE_MODULE_PATH "${ORIGINAL_SOURCE}/cmake/modules")
set(ARCH x86_64)
set(ENABLE_RUST_USER_TOOLS ON CACHE BOOL "Require selected Rust user tools for this qualification")
if(NOT ENABLE_RUST_USER_TOOLS)
  message(FATAL_ERROR "This qualification requires ENABLE_RUST_USER_TOOLS=ON")
endif()
set(ENABLE_QLMPI OFF)
set(ENABLE_UTI OFF)
set(LIBDWARF "LIBDWARF-NOTFOUND")
set(LIBBFD "/usr/lib64/libbfd.so")
set(RUSTC "/nightly/bin/rustc")
set(MCKERNEL_RUNTIME_TOOL_RUSTC "/nightly/bin/rustc" CACHE FILEPATH "Pinned runtime compiler for current user-tool configuration")
find_library(LIBNUMA numa REQUIRED)
find_library(LIBRT rt REQUIRED)
set(IHK_FULL_SOURCE_DIR "${ORIGINAL_SOURCE}/ihk")
set(KERNEL_DIR "/out/build")
include(GNUInstallDirs)
add_subdirectory("${STAGED_USER_SOURCE}" executer/user)
