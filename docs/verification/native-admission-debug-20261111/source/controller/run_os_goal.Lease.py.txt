class Lease:
    def __init__(self, path):
        self.file = path.open("a+")
        try:
            fcntl.flock(self.file.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
        except BlockingIOError:
            self.file.close()
            raise RuntimeError("Another OS launcher holds the repository lease.")

    def close(self):
        self.file.close()
